\begin{question}[class=Z]{10}
  You want to start a savings account for a graduation trip that you wish to take in \SavingsTerm~years.
  Your plan is to make an initial deposit of \(\$\Principal\) and then recurring deposits of \(\$\Deposits\) each month into an account paying interest at a rate of \(\Rate\%\) per year compounded monthly.
  How much will you have saved in the account at the end of the \SavingsTerm~years?\\
  \FormulaInstructions~\RoundUnitsInstructions
  \begin{tasks}(7)
    \task[\selectall] \ A
    \task[\selectall] \ B
    \task[\correctselectall] \ C
    \task[\selectall] \ D
    \task[\selectall] \ E
    \task[\correctselectall] \ F
    \task[\selectall] \ G
  \end{tasks}
  \mrubric{1 pt \\ each}
\end{question}
\begin{minipage}{0.70\textwidth}
  \begin{solution}
    \begin{align*}
      A = \Principal \left( 1 + \dfrac{\DecimalRate}{12} \right)^{\SavingsTerm\cdot12} &= \Accumulated \\
      S = \Deposits \left[ \dfrac{\left(1 + \frac{\DecimalRate}{12} \right)^{\SavingsTerm\cdot12} - 1}{\frac{\DecimalRate}{12}} \right] &= \FutureValue \\
      \text{Total } = A + S &= \TotalSavings
    \end{align*}
  \end{solution}
  \rubric{3 pts each formula and 2 pts final answer}
\end{minipage}\hspace{\fill}%
\begin{minipage}{0.25\textwidth}
  \vspace{1.5in}
  \AnswerBox{Final Answer:}{\(\$\TotalSavings\)}
  \vspace{0.1in}
\end{minipage}
