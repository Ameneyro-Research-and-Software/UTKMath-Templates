\begin{question}[class=Z]
  Consider the following system of linear equations.
  \[
    \begin{cases}
      -2x + y = 1 \\
      x + y = 1
    \end{cases}
  \]
  \begin{enumerate}[label = \textbf{\alph*)}]
    \item{} [\addpoints{4}] Find the intercepts of each line and write them as points in the corresponding cells.
          \begin{center}
            \intercepts{(-1/2,0)}{(0,1)}{(1,0)}{(0,1)}
          \end{center}
    \item{} [\addpoints{3}] \MultipleChoiceInstructions \\[0.05in]
          Which graph shows the lines in the given system of equations?
          \begin{tasks}(4)
            \task[\choice] \ \(\begin{smallgraph} \addplot {-x + 1}; \addplot {x + 2}; \end{smallgraph}\)
            \task[\choice] \ \(\begin{smallgraph} \addplot {x - 1}; \addplot {2*x + 1}; \end{smallgraph}\)
            \task[\correctchoice] \ \(\begin{smallgraph} \addplot {-x + 1}; \addplot {2*x + 1}; \end{smallgraph}\)
            \task[\choice] \ \(\begin{smallgraph} \addplot {-x + 1}; \addplot {-2*x - 1}; \end{smallgraph}\)
          \end{tasks}
    \item{} [\addpoints{2}] \MultipleChoiceInstructions \\[0.05in]
          What is the solution to the system of equations?
          \begin{tasks}(4)
            \task[\choice] \ \(\left(-\dfrac{1}{2}, \dfrac{3}{2}\right)\)
            \task[\choice] \ \((-2,-3)\)
            \task[\correctchoice] \ \((0,1)\)
            \task[\choice] \ \((-2,3)\)
          \end{tasks}
  \end{enumerate}
\end{question}
